05 快速上手

// zvec_example.py — pip install zvec 后直接跑 
import zvec

# 定义 Collection schema
schema = zvec.CollectionSchema(
    name="example",
    vectors=zvec.VectorSchema("embedding", zvec.DataType.VECTOR_FP32, 4),
)

# 创建并打开 collection（存本地 ./zvec_example 目录）
collection = zvec.create_and_open(path="./zvec_example", schema=schema)

# 插入文档向量
collection.insert([
    zvec.Doc(id="doc_1", vectors={"embedding": [0.1, 0.2, 0.3, 0.4]}),
    zvec.Doc(id="doc_2", vectors={"embedding": [0.2, 0.3, 0.4, 0.1]}),
])

# 向量相似性搜索
results = collection.query(
    zvec.VectorQuery("embedding", vector=[0.4, 0.3, 0.3, 0.1]),
    topk=10
)
print(results)  # [{'id': 'doc_2', 'score': 0.94, ...}, ...]
// 混合检索（FTS + 向量）— v0.5.0 新特性 
# 全文本 + 向量 + 标量过滤，一次 MultiQuery 搞定
results = collection.query(
    zvec.MultiQuery(
        vector_query=zvec.VectorQuery("embedding", vector=[0.4, 0.3, 0.3, 0.1]),
        fts_query="zvec OR database",
        filter="category == 'tech'",
    ),
    topk=10
)

06 v0.5.0 新特性（2026-06-12）
Full-Text Search

原生全文检索 FTS，无需外部搜索引擎，字符串字段直接建索引，支持自然语言查询。

Hybrid Retrieval

MultiQuery 融合向量 + FTS + 标量过滤，一句 query 搞定混合搜索精度。

DiskANN Index

磁盘驻留索引，内存不够也能跑大模型知识库，资源消耗大幅降低。

Go / Rust SDK

官方新增 Go / Rust 高性能绑定，Zvec Studio 可视化工具，RISC-V 平台支持。
