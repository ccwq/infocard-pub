CORE CODE PATTERNS · 核心代码模式
① Timeline · 时间线const tl = gsap.timeline({ defaults: { duration: 0.6, ease: "power2.out" } }); tl.to(".hero", { x: 100, opacity: 1 }) .to(".sub", { y: 20 }, "+=0.3") // 相对加 0.3s .to(".btn", { scale: 1 }, "-=0.15") // 相对减 .to(".bg", { blur: 10 }, "<") // 对齐前一个开始
② ScrollTrigger · 滚动驱动gsap.registerPlugin(ScrollTrigger); // scrub 绑定滚动位置 gsap.to(".panel", { x: 200, scrollTrigger: { trigger: ".section", start: "top center", end: "bottom top", scrub: true, pin: true, // 固定元素 } }); // 布局变化后刷新：ScrollTrigger.refresh();
③ React · useGSAPimport { useGSAP } from "@gsap/react"; import { gsap } from "gsap"; useGSAP(() => { gsap.from(ref.current, { x: -50, opacity: 0 }); }, { scope: containerRef }); // ✅ 必须在 unmount 时 cleanup return () => ctx.revert();
④ Performance · 性能原则// ✅ 正确：使用 transform（GPU 加速） gsap.to(el, { x: 200, rotation: 15, scale: 1.1 }); // ❌ 错误：layout props 触发强制重排（reflow） gsap.to(el, { left: 100, width: 300 });
